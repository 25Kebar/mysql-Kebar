N°9  Muestre en una sola fila el salario más alto (mayor),el más bajo(menor)y el promedio redondeado a 2 decimales(promedio).

SELECT MAX(salario) AS salario_mas_alto, MIN(salario) AS salario_mas_bajo, ROUND(AVG(salario),2) AS promedio FROM empleados;

N°10 ¿Cuánto suma la nómina completa de la empresa? Nombre la columna nomina_total.

SELECT SUM(salario) AS nomina_total FROM empleados;

SELECT
    d.nombre                  AS departamento,
    COUNT(DISTINCT s.id)      AS sedes_con_personal,
    COUNT(e.id)               AS empleados,
    SUM(e.salario)            AS nomina
FROM departamentos d
INNER JOIN sedes s     ON s.id_departamento = d.id
INNER JOIN empleados e ON e.sede_id = s.id
WHERE e.fecha_ingreso >= '2022-01-01'
GROUP BY d.id, d.nombre
HAVING SUM(e.salario) > 8000000
ORDER BY nomina DESC;
