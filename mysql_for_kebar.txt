-- Ejercicio 1: creo la tabla de empleados
CREATE TABLE empleados (
    id            INT           NOT NULL AUTO_INCREMENT,
    sede_id       INT           NOT NULL,
    nombre        VARCHAR(80)   NOT NULL,
    cargo         VARCHAR(50)   DEFAULT NULL,
    salario       DECIMAL(10,2) NOT NULL,
    fecha_ingreso DATE          DEFAULT NULL,
    PRIMARY KEY (id),
    CONSTRAINT fk_empleados_sedes FOREIGN KEY (sede_id) REFERENCES sedes (id) ON DELETE CASCADE,
    CONSTRAINT chk_empleados_salario CHECK (salario > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

    SELECT 1

SELECT
    e.id,
    e.nombre,
    e.sede_id,
    e.salario,
    (
        SELECT AVG(e2.salario)
        FROM empleados AS e2
        WHERE e2.sede_id = e.sede_id
    ) AS promedio_sede
FROM empleados AS e
WHERE e.salario > (
    SELECT AVG(e2.salario)
    FROM empleados AS e2
    WHERE e2.sede_id = e.sede_id
)
ORDER BY e.sede_id ASC, e.id ASC;
-- Ejercicio 5
SELECT n.sede_id, n.nomina
FROM (
    SELECT sede_id, SUM(salario) AS nomina
    FROM empleados
    GROUP BY sede_id
) AS n
WHERE n.nomina > (
    SELECT AVG(n2.nomina)
    FROM (
        SELECT sede_id, SUM(salario) AS nomina
        FROM empleados
        GROUP BY sede_id
    ) AS n2
)
ORDER BY n.nomina DESC, n.sede_id ASC;

CREATE OR REPLACE VIEW vw_s13_salarios_superiores_promedio

    FROM empleados
);

FROM vw_s13_salarios_superiores_promedio

DELIMITER $$

CREATE PROCEDURE empleados_por_cargo(
    IN cargo_buscado VARCHAR(50)
)
BEGIN
    SELECT
        e.id,
        e.nombre AS empleado,
        e.cargo,
        s.nombre AS sede
    FROM empleados AS e
    INNER JOIN sedes AS s
        ON e.sede_id = s.id
    WHERE e.cargo = cargo_buscado;
END $$

DELIMITER ;

CREATE PROCEDURE resumen_sede(
    IN id_sede INT,
    OUT total INT,
    OUT nomina DECIMAL(12,2),
    OUT promedio DECIMAL(12,2)
)
BEGIN

    IF NOT EXISTS (
        SELECT 1
        FROM sedes
        WHERE id = id_sede
    ) THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Error: la sede no existe';

    ELSE

        SELECT
            COUNT(*),
            COALESCE(SUM(salario), 0),
            COALESCE(AVG(salario), 0)
        INTO total, nomina, promedio
        FROM empleados
        WHERE sede_id = id_sede;

    END IF;

END //

DELIMITER ;
